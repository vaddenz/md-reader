**Goal:** 在 md-reader v1.0.1 基础上增量交付四个特性：阅读排版设置（字体/字号/表格与引用比例）、内置字体、持久化文字高亮与自定义右键菜单、手动横屏锁定。

**Tech Stack:** React 18 + TS + Vite 6 + IndexedDB(idb) + Vitest + Playwright；新增 devDependency 仅 `wawoff2`（字体脚本用）。

**Spec:** `docs/specs/2026-10-18-md-reader-v1.1-design.md`（本计划逐条实现该 spec；冲突以 spec 为准）

```ts
import type { CSSProperties } from 'react';
import { kvGet, kvSet } from './db';
import { serialize } from './serialize';

export type FontChoice = 'system' | 'xihei' | 'serif';

export interface ReaderAppearance {
  fontFamily: FontChoice;
  fontSize: number;   // px
  tableScale: number; // 相对正文比例
  quoteScale: number;
}

export const FONT_SIZE_RANGE = { min: 12, max: 24 } as const;
export const SCALE_RANGE = { min: 0.5, max: 1.2 } as const;

export const DEFAULT_APPEARANCE: ReaderAppearance = {
  fontFamily: 'system', fontSize: 17, tableScale: 0.95, quoteScale: 1,
};

export const FONT_STACKS: Record<FontChoice, string> = {
  system: 'var(--font-sans)',
  xihei: "'LXGW Neo XiHei Screen', var(--font-sans)",
  serif: "'Noto Serif SC', serif",
};

const APPEARANCE_KEY = 'appearance';

const clamp = (v: number, min: number, max: number) => Math.min(max, Math.max(min, v));

export function sanitizeAppearance(raw: unknown): ReaderAppearance {
  if (!raw || typeof raw !== 'object' || Array.isArray(raw)) return { ...DEFAULT_APPEARANCE };
  const rec = raw as Partial<ReaderAppearance>;
  return {
    fontFamily: rec.fontFamily === 'xihei' || rec.fontFamily === 'serif' ? rec.fontFamily : 'system',
    fontSize: Number.isFinite(rec.fontSize) ? clamp(rec.fontSize as number, FONT_SIZE_RANGE.min, FONT_SIZE_RANGE.max) : DEFAULT_APPEARANCE.fontSize,
    tableScale: Number.isFinite(rec.tableScale) ? clamp(rec.tableScale as number, SCALE_RANGE.min, SCALE_RANGE.max) : DEFAULT_APPEARANCE.tableScale,
    quoteScale: Number.isFinite(rec.quoteScale) ? clamp(rec.quoteScale as number, SCALE_RANGE.min, SCALE_RANGE.max) : DEFAULT_APPEARANCE.quoteScale,
  };
}

export async function getAppearance(): Promise<ReaderAppearance> {
  return sanitizeAppearance(await kvGet<unknown>(APPEARANCE_KEY));
}

export function saveAppearance(a: ReaderAppearance): Promise<void> {
  const clean = sanitizeAppearance(a);
  return serialize(async () => { await kvSet(APPEARANCE_KEY, clean); });
}

export function appearanceStyle(a: ReaderAppearance): CSSProperties {
  return {
    '--r-font-family': FONT_STACKS[a.fontFamily],
    '--r-font-size': `${a.fontSize}px`,
    '--r-table-scale': String(a.tableScale),
    '--r-quote-scale': String(a.quoteScale),
  } as CSSProperties;
}
```

```ts
/** 写入 kv 键（测试预置数据用）。 */
export async function writeKv(page: Page, key: string, value: unknown): Promise<void> {
  const ok = await page.evaluate(
    (args) =>
      new Promise<boolean>((resolve) => {
        const req = indexedDB.open('mdreader');
        req.onupgradeneeded = () => {
          const db = req.result;
          for (const name of ['files', 'blobs', 'kv']) {
            if (!db.objectStoreNames.contains(name)) db.createObjectStore(name);
          }
        };
        req.onerror = () => resolve(false);
        req.onsuccess = () => {
          const db = req.result;
          try {
            const put = db.transaction('kv', 'readwrite').objectStore('kv').put(args.value, args.key);
            put.onsuccess = () => { resolve(true); db.close(); };
            put.onerror = () => { resolve(false); db.close(); };
          } catch { resolve(false); db.close(); }
        };
      }),
    { key, value },
  );
  if (!ok) throw new Error(`writeKv 失败：${key}`);
}

/** 读取 kv 键；不存在或失败返回 undefined。 */
export async function readKv(page: Page, key: string): Promise<unknown> {
  return page.evaluate(
    (k) =>
      new Promise<unknown>((resolve) => {
        const req = indexedDB.open('mdreader');
        req.onupgradeneeded = () => {
          const db = req.result;
          for (const name of ['files', 'blobs', 'kv']) {
            if (!db.objectStoreNames.contains(name)) db.createObjectStore(name);
          }
        };
        req.onerror = () => resolve(undefined);
        req.onsuccess = () => {
          const db = req.result;
          try {
            const get = db.transaction('kv', 'readonly').objectStore('kv').get(k);
            get.onsuccess = () => { resolve(get.result); db.close(); };
            get.onerror = () => { resolve(undefined); db.close(); };
          } catch { resolve(undefined); db.close(); }
        };
      }),
    key,
  );
}
```

```tsx
import { useEffect, useRef, useState } from 'react';
import {
  appearanceStyle, DEFAULT_APPEARANCE, FONT_SIZE_RANGE, SCALE_RANGE,
  getAppearance, saveAppearance, type FontChoice, type ReaderAppearance,
} from '../data/appearance';
import { showToast } from '../lib/toast';

const FONT_OPTIONS: { id: FontChoice; label: string }[] = [
  { id: 'system', label: '系统默认' },
  { id: 'xihei', label: '等线（霞鹜新晰黑）' },
  { id: 'serif', label: '宋体（思源宋体）' },
];

/** 设置页「阅读外观」区：字体 / 正文字号 / 表格与引用比例 + 实时预览（spec §3.3）。 */
export default function AppearanceSection() {
  const [a, setA] = useState<ReaderAppearance>(DEFAULT_APPEARANCE);
  const timer = useRef(0);
  const pending = useRef<ReaderAppearance | null>(null);

  useEffect(() => {
    let cancelled = false;
    void getAppearance().then((v) => { if (!cancelled) setA(v); });
    return () => { cancelled = true; };
  }, []);

  const flush = () => {
    const v = pending.current;
    if (!v) return;
    pending.current = null;
    void saveAppearance(v).catch(() => showToast('外观设置保存失败', { kind: 'error' }));
  };
  // 卸载时冲刷未落盘的防抖写（否则 300ms 内离开页面会丢最后一次修改）。
  useEffect(() => () => { window.clearTimeout(timer.current); flush(); }, []);

  const update = (next: ReaderAppearance) => {
    setA(next);
    pending.current = next;
    window.clearTimeout(timer.current);
    timer.current = window.setTimeout(flush, 300);
  };

  const pct = (v: number) => `${Math.round(v * 100)}%`;

  return (
    <section className="settings-section" aria-labelledby="settings-appearance-title">
      <h2 id="settings-appearance-title">阅读外观</h2>
      <div className="appearance-row" role="radiogroup" aria-label="正文字体">
        {FONT_OPTIONS.map((o) => (
          <label key={o.id} className="appearance-radio">
            <input
              type="radio" name="appearance-font" checked={a.fontFamily === o.id}
              onChange={() => update({ ...a, fontFamily: o.id })}
            />
            {o.label}
          </label>
        ))}
      </div>
      <label className="appearance-slider">
        正文字号 <output>{a.fontSize}px</output>
        <input type="range" min={FONT_SIZE_RANGE.min} max={FONT_SIZE_RANGE.max} step={1} value={a.fontSize}
          onChange={(e) => update({ ...a, fontSize: Number(e.target.value) })} />
      </label>
      <label className="appearance-slider">
        表格字号比例 <output>{pct(a.tableScale)}</output>
        <input type="range" min={SCALE_RANGE.min} max={SCALE_RANGE.max} step={0.05} value={a.tableScale}
          onChange={(e) => update({ ...a, tableScale: Number(e.target.value) })} />
      </label>
      <label className="appearance-slider">
        引用字号比例 <output>{pct(a.quoteScale)}</output>
```

```ts
import { kvGet, kvSet } from './db';
import { serialize } from './serialize';

export interface Decoration { kind: 'bg'; color: string }
export interface HighlightRecord {
  id: string; exact: string; prefix: string; suffix: string;
  decorations: Decoration[]; createdAt: number;
}
export type HighlightMap = Record<string, HighlightRecord[]>;

/** 颜色注册表（渲染与菜单共用；加新颜色 = 加一项 + 一个 CSS 变量）。 */
export const HIGHLIGHT_COLORS = [
  { id: 'yellow', label: '高亮·黄', cssVar: '--hl-yellow' },
  { id: 'red', label: '高亮·红', cssVar: '--hl-red' },
] as const;

export const HIGHLIGHT_FORMAT_VERSION = 1;
const HIGHLIGHTS_KEY = 'highlights';

function sanitizeRecord(raw: unknown): HighlightRecord | null {
  if (!raw || typeof raw !== 'object' || Array.isArray(raw)) return null;
  const rec = raw as Partial<HighlightRecord>;
  if (typeof rec.id !== 'string' || !rec.id) return null;
  if (typeof rec.exact !== 'string' || !rec.exact) return null;
  const decorations: Decoration[] = Array.isArray(rec.decorations)
    ? rec.decorations
        .filter((d): d is Decoration =>
          !!d && typeof d === 'object' && (d as Decoration).kind === 'bg' && typeof (d as Decoration).color === 'string')
        .map((d) => ({ kind: 'bg', color: d.color }))
    : [];
  return {
    id: rec.id, exact: rec.exact,
    prefix: typeof rec.prefix === 'string' ? rec.prefix : '',
    suffix: typeof rec.suffix === 'string' ? rec.suffix : '',
    decorations,
    createdAt: typeof rec.createdAt === 'number' ? rec.createdAt : 0,
  };
}

export function sanitizeHighlights(raw: unknown): HighlightMap {
  if (!raw || typeof raw !== 'object' || Array.isArray(raw)) return {};
  const out: HighlightMap = {};
  for (const [path, list] of Object.entries(raw as Record<string, unknown>)) {
    if (!Array.isArray(list)) continue;
    const records = list.map(sanitizeRecord).filter((r): r is HighlightRecord => r !== null);
    if (records.length) out[path] = records;
  }
  return out;
}

async function readAll(): Promise<HighlightMap> {
  return sanitizeHighlights(await kvGet<unknown>(HIGHLIGHTS_KEY));
}

export async function getHighlights(): Promise<HighlightMap> { return readAll(); }

export async function getDocHighlights(path: string): Promise<HighlightRecord[]> {
  return (await readAll())[path] ?? [];
}

export function addHighlight(path: string, rec: HighlightRecord): Promise<void> {
  return serialize(async () => {
    const all = await readAll();
    all[path] = [...(all[path] ?? []), rec];
    await kvSet(HIGHLIGHTS_KEY, all);
  });
}

export function removeHighlights(path: string, ids: string[]): Promise<void> {
  return serialize(async () => {
    const all = await readAll();
    const rest = (all[path] ?? []).filter((r) => !ids.includes(r.id));
    if (rest.length) all[path] = rest; else delete all[path];
    await kvSet(HIGHLIGHTS_KEY, all);
  });
}

/** 导出（含孤儿记录）；格式带版本号。 */
export async function exportHighlights(): Promise<string> {
  return JSON.stringify({ formatVersion: HIGHLIGHT_FORMAT_VERSION, exportedAt: Date.now(), highlights: await readAll() }, null, 2);
}

/** 导入合并：键 = path + exact + decorations；重复跳过、不覆盖本地已有。 */
export function importHighlights(json: string): Promise<{ imported: number; skipped: number }> {
  return serialize(async () => {
    const parsed = JSON.parse(json) as { formatVersion?: unknown; highlights?: unknown };
    if (parsed?.formatVersion !== HIGHLIGHT_FORMAT_VERSION) throw new Error('不支持的格式版本');
    const incoming = sanitizeHighlights(parsed.highlights);
    const all = await readAll();
    let imported = 0; let skipped = 0;
    const key = (p: string, r: HighlightRecord) => `${p}\u0000${r.exact}\u0000${JSON.stringify(r.decorations)}`;
    for (const [path, list] of Object.entries(incoming)) {
      const existing = new Set((all[path] ?? []).map((r) => key(path, r)));
      for (const r of list) {
        if (existing.has(key(path, r))) { skipped += 1; continue; }
        existing.add(key(path, r));
        all[path] = [...(all[path] ?? []), r];
        imported += 1;
      }
    }
    await kvSet(HIGHLIGHTS_KEY, all);
    return { imported, skipped };
  });
}
```

参考实现（要点：跳过清单、块边界 `\n`、归一化匹配 + 上下文计分、range 映射包裹、冲突先到先得）：

```ts
import type { HighlightRecord } from '../data/highlights';

export interface TextPos { node: Text | null; offset: number }
export interface Haystack { text: string; map: TextPos[] }

const SKIP_SELECTOR = 'script, style, pre, code, button, svg, .code-copy, .mermaid, .md-img-error';
const BLOCK_SELECTOR = 'p, li, h1, h2, h3, h4, h5, h6, blockquote, td, th, dt, dd';
const CONTEXT = 32;

export function buildHaystack(container: HTMLElement): Haystack {
  const text: string[] = [];
  const map: TextPos[] = [];
  const walker = document.createTreeWalker(container, NodeFilter.SHOW_TEXT);
  let prevBlock: Element | null = null;
  let node = walker.nextNode() as Text | null;
  while (node) {
    const parent = node.parentElement;
    if (parent && !parent.closest(SKIP_SELECTOR)) {
      const block = parent.closest(BLOCK_SELECTOR) ?? container;
      if (prevBlock !== null && block !== prevBlock) {
        text.push('\n');
        map.push({ node: null, offset: -1 });
      }
      prevBlock = block;
      for (let i = 0; i < node.data.length; i++) {
        text.push(node.data[i]);
        map.push({ node, offset: i });
      }
    }
    node = walker.nextNode() as Text | null;
  }
  return { text: text.join(''), map };
}

/** 归一化：空白串折叠为单空格，并记录每个归一化字符 → 原文下标。 */
function normalize(s: string): { norm: string; toRaw: number[] } {
  const norm: string[] = [];
  const toRaw: number[] = [];
  let i = 0;
  while (i < s.length) {
    if (/\s/.test(s[i])) {
      norm.push(' ');
      toRaw.push(i);
      while (i < s.length && /\s/.test(s[i])) i++;
    } else {
      norm.push(s[i]);
      toRaw.push(i);
      i++;
    }
  }
  return { norm: norm.join(''), toRaw };
}

function commonSuffixLen(a: string, b: string): number {
  let n = 0;
  while (n < a.length && n < b.length && a[a.length - 1 - n] === b[b.length - 1 - n]) n++;
  return n;
}
function commonPrefixLen(a: string, b: string): number {
  let n = 0;
  while (n < a.length && n < b.length && a[n] === b[n]) n++;
  return n;
}

export function locateAnchor(haystack: Haystack, anchor: { exact: string; prefix: string; suffix: string }): { start: number; end: number } | null {
  // 一级：原始精确匹配（候选直接带原文区间）
  const candidates: { start: number; end: number }[] = [];
  for (let i = haystack.text.indexOf(anchor.exact); i !== -1; i = haystack.text.indexOf(anchor.exact, i + 1)) {
    candidates.push({ start: i, end: i + anchor.exact.length });
  }
  // 二级：空白归一化匹配（toRaw 映射回原文；end 取下一归一化字符的原文下标，覆盖整段空白）
  let normInfo: { norm: string; toRaw: number[] } | null = null;
  let nExact = anchor.exact;
  if (candidates.length === 0) {
    normInfo = normalize(haystack.text);
    nExact = normalize(anchor.exact).norm;
    if (!nExact) return null;
    for (let i = normInfo.norm.indexOf(nExact); i !== -1; i = normInfo.norm.indexOf(nExact, i + 1)) {
      const start = normInfo.toRaw[i];
      const after = i + nExact.length;
      const end = after < normInfo.toRaw.length ? normInfo.toRaw[after] : haystack.text.length;
      candidates.push({ start, end });
    }
  }
  if (candidates.length === 0) return null;
  // 三级：上下文消歧（归一化空间计分；平分取最早）
  const useNorm = normInfo ?? normalize(haystack.text);
  const useExact = normInfo ? nExact : normalize(anchor.exact).norm;
  const nPrefix = normalize(anchor.prefix).norm;
  const nSuffix = normalize(anchor.suffix).norm;
  const rawToNorm = new Map<number, number>(); // 原文下标 → 归一化下标（消歧用）
  useNorm.toRaw.forEach((raw, i) => { if (!rawToNorm.has(raw)) rawToNorm.set(raw, i); });
  let best = { range: candidates[0], score: -1 };
  for (const range of candidates) {
    const nStart = rawToNorm.get(range.start) ?? 0;
    const nEnd = nStart + useExact.length;
    const before = useNorm.norm.slice(Math.max(0, nStart - CONTEXT), nStart);
    const after = useNorm.norm.slice(nEnd, nEnd + CONTEXT);
    const score = commonSuffixLen(before, nPrefix) + commonPrefixLen(after, nSuffix);
    if (score > best.score) best = { range, score };
  }
  return best.range;
}
```

（说明：`normalize` 折叠空白并保留「归一化下标 → 原文下标」映射；消歧在归一化空间计分，取公共前后缀匹配长度最高者，平分取最早位置。）

```ts
export function markClassFor(record: HighlightRecord): string | null {
  const known = record.decorations.filter((d) => d.kind === 'bg' && (d.color === 'yellow' || d.color === 'red'));
  if (known.length === 0) return null;
  return `hl hl-${known[0].color}`;
}

export function wrapRange(container: HTMLElement, haystack: Haystack, start: number, end: number, className: string, id: string): HTMLElement[] {
  const groups: { node: Text; from: number; to: number }[] = [];
  for (let i = start; i < end; i++) {
    const pos = haystack.map[i];
    if (!pos || pos.node === null) continue; // 分隔符
    const last = groups[groups.length - 1];
    if (last && last.node === pos.node && pos.offset === last.to) last.to = pos.offset + 1;
    else groups.push({ node: pos.node, from: pos.offset, to: pos.offset + 1 });
  }
  const marks: HTMLElement[] = [];
  for (const g of groups) {
    const range = document.createRange();
    range.setStart(g.node, g.from);
    range.setEnd(g.node, g.to);
    const mark = document.createElement('mark');
    mark.className = className;
    mark.setAttribute('data-hl-id', id);
    range.surroundContents(mark);
    marks.push(mark);
  }
  return marks;
}

export function removeMarks(container: HTMLElement, ids: string[]): void {
  for (const mark of container.querySelectorAll('mark.hl')) {
    const id = mark.getAttribute('data-hl-id');
    if (id && ids.includes(id)) mark.replaceWith(...mark.childNodes);
  }
  container.normalize();
}

export function applyHighlights(container: HTMLElement, records: HighlightRecord[]): Map<string, { start: number; end: number }> {
  const applied = new Map<string, { start: number; end: number }>();
  const sorted = [...records].sort((a, b) => a.createdAt - b.createdAt);
  const haystack = buildHaystack(container);
  for (const record of sorted) {
    const cls = markClassFor(record);
    if (!cls) continue;
    const found = locateAnchor(haystack, record);
    if (!found) continue; // 孤儿：保留记录不渲染
    let conflict = false;
    for (const r of applied.values()) if (found.start < r.end && r.start < found.end) { conflict = true; break; }
    if (conflict) continue; // 先到先得
    wrapRange(container, haystack, found.start, found.end, cls, record.id);
    applied.set(record.id, found);
  }
  return applied;
}

export function rangeIntersectsExcluded(container: HTMLElement, range: Range): boolean {
  for (const el of container.querySelectorAll(SKIP_SELECTOR)) {
    if (range.intersectsNode(el)) return true;
  }
  return false;
}

export function selectionToAnchor(container: HTMLElement, range: Range): { anchor: { exact: string; prefix: string; suffix: string }; start: number; end: number } | null {
  if (rangeIntersectsExcluded(container, range)) return null;
  const haystack = buildHaystack(container);
  let start = -1; let end = -1;
  for (let i = 0; i < haystack.map.length; i++) {
    const pos = haystack.map[i];
    if (!pos.node) continue;
    const a = range.comparePoint(pos.node, pos.offset);
    const b = range.comparePoint(pos.node, pos.offset + 1);
    if (a >= 0 && b <= 0) {
      if (start === -1) start = i;
      end = i + 1;
    }
  }
  if (start === -1 || start >= end) return null;
  return {
    anchor: {
      exact: haystack.text.slice(start, end),
      prefix: haystack.text.slice(Math.max(0, start - CONTEXT), start),
      suffix: haystack.text.slice(end, end + CONTEXT),
    },
    start, end,
  };
}
```

```ts
import { HIGHLIGHT_COLORS } from '../data/highlights';

export interface MenuContext {
  hasSelection: boolean;
  selectionInExcluded: boolean;
  selectionIntersectsHighlight: boolean;
  targetOnHighlight: boolean;
}

/** 菜单项判定（spec §6.2 表格；纯函数，单测见 tests/unit/selection-menu.test.ts）。 */
export function menuActionsFor(ctx: MenuContext): MenuAction[] {
  if (ctx.hasSelection) {
    if (ctx.selectionInExcluded) return [{ id: 'copy', label: '复制' }];
    const actions: MenuAction[] = [{ id: 'copy', label: '复制' }];
    for (const c of HIGHLIGHT_COLORS) actions.push({ id: `hl:${c.id}`, label: c.label });
    if (ctx.selectionIntersectsHighlight) actions.push({ id: 'clear', label: '清除高亮' });
    return actions;
  }
  const actions: MenuAction[] = [];
  if (ctx.targetOnHighlight) actions.push({ id: 'clear', label: '清除高亮' });
  actions.push({ id: 'select-all', label: '全选' });
  return actions;
}
```

```tsx
import { useEffect, useLayoutEffect, useRef, useState } from 'react';
import type { MenuAction } from '../lib/selection-menu';

interface Props {
  x: number; y: number;
  actions: MenuAction[];
  onAction: (id: string) => void;
  onClose: () => void;
}

/** 阅读正文自定义右键/长按菜单（spec §6.3）：触点定位、越界收边、外部/滚动/Esc 关闭。 */
export default function SelectionMenu({ x, y, actions, onAction, onClose }: Props) {
  const ref = useRef<HTMLDivElement>(null);
  const [pos, setPos] = useState({ left: x, top: y });

  useLayoutEffect(() => {
    const el = ref.current;
    if (!el) return;
    const r = el.getBoundingClientRect();
    setPos({
      left: Math.max(8, Math.min(x, window.innerWidth - r.width - 8)),
      top: Math.max(8, Math.min(y, window.innerHeight - r.height - 8)),
    });
  }, [x, y]);

  useEffect(() => {
    const onDown = (e: PointerEvent) => { if (!ref.current?.contains(e.target as Node)) onClose(); };
    const onKey = (e: KeyboardEvent) => { if (e.key === 'Escape') onClose(); };
    const onScroll = () => onClose();
    document.addEventListener('pointerdown', onDown, true);
    document.addEventListener('keydown', onKey);
    window.addEventListener('scroll', onScroll, { passive: true });
    return () => {
      document.removeEventListener('pointerdown', onDown, true);
      document.removeEventListener('keydown', onKey);
      window.removeEventListener('scroll', onScroll);
    };
  }, [onClose]);

  return (
    <div ref={ref} className="selection-menu" style={{ left: pos.left, top: pos.top }} role="menu">
      {actions.map((a) => (
        <button key={a.id} type="button" role="menuitem" className="selection-menu-item" onClick={() => onAction(a.id)}>
          {a.label}
        </button>
      ))}
    </div>
  );
}
```

```tsx
import SelectionMenu from '../components/SelectionMenu';
import { menuActionsFor, type MenuContext } from '../lib/selection-menu';
import { addHighlight, removeHighlights, type HighlightRecord } from '../data/highlights';
import { buildHaystack, rangeIntersectsExcluded, removeMarks, selectionToAnchor, wrapRange } from '../markdown/highlights';
// …组件内：
interface MenuState {
  x: number; y: number;
  ctx: MenuContext;
  sel: { anchor: { exact: string; prefix: string; suffix: string }; start: number; end: number } | null;
  markId: string | null; // 无选区时点中的高亮记录 id（「清除高亮」用）
}
const [menu, setMenu] = useState<MenuState | null>(null);

const onContentContextMenu = useCallback((e: React.MouseEvent<HTMLDivElement>) => {
  e.preventDefault();
  const container = contentRef.current;
  if (!container) return;
  const sel = window.getSelection();
  const range = sel && sel.rangeCount > 0 && !sel.isCollapsed ? sel.getRangeAt(0) : null;
  const inContainer = !!range && container.contains(range.commonAncestorContainer);
  const excluded = !!range && inContainer && rangeIntersectsExcluded(container, range);
  const selInfo = range && inContainer && !excluded ? selectionToAnchor(container, range) : null;
  let intersects = false;
  if (selInfo) {
    for (const r of appliedRef.current.values()) {
      if (selInfo.start < r.end && r.start < selInfo.end) { intersects = true; break; }
    }
  }
  const markEl = (e.target as Element).closest('mark.hl');
  setMenu({
    x: e.clientX, y: e.clientY,
    ctx: {
      hasSelection: !!range && inContainer, // 有选区（含排除区；能否高亮由 selectionInExcluded 决定）
      selectionInExcluded: excluded,
      selectionIntersectsHighlight: intersects,
      targetOnHighlight: !!markEl,
    },
    sel: selInfo,
    markId: markEl?.getAttribute('data-hl-id') ?? null,
  });
}, []);

const onMenuAction = useCallback((id: string) => {
  const current = menu;
  setMenu(null);
  const container = contentRef.current;
  if (!current || !container) return;
  const sel = window.getSelection();

  if (id === 'copy') {
    const text = sel?.toString() ?? '';
    void navigator.clipboard.writeText(text).then(
      () => showToast('已复制'),
      () => showToast('复制失败', { kind: 'error' }),
    );
    return;
  }
  if (id === 'select-all') {
    const r = document.createRange();
    r.selectNodeContents(container);
    sel?.removeAllRanges();
    sel?.addRange(r);
    return;
  }
  if (id === 'clear') {
    const ids: string[] = [];
    if (current.sel) {
      for (const [recId, r] of appliedRef.current) {
        if (current.sel.start < r.end && r.start < current.sel.end) ids.push(recId);
      }
    } else if (current.markId) {
      ids.push(current.markId); // 无选区、点在高亮上
    }
    if (ids.length) {
      removeMarks(container, ids);
      for (const rid of ids) appliedRef.current.delete(rid);
      void removeHighlights(path, ids).catch(() => {});
    }
    sel?.removeAllRanges();
    return;
  }
  if (id.startsWith('hl:')) {
    const color = id.slice(3);
    if (!current.sel) return;
    const container2 = contentRef.current;
    if (!container2) return;
    // 1) 清除与选区相交的已渲染高亮
    const overlapIds: string[] = [];
    for (const [recId, r] of appliedRef.current) {
      if (current.sel.start < r.end && r.start < current.sel.end) overlapIds.push(recId);
    }
    if (overlapIds.length) {
      removeMarks(container2, overlapIds);
      for (const rid of overlapIds) appliedRef.current.delete(rid);
      void removeHighlights(path, overlapIds).catch(() => {});
    }
    // 2) 包裹新高亮（DOM 变化后重建 haystack）
    const haystack = buildHaystack(container2);
    const record: HighlightRecord = {
      id: crypto.randomUUID(),
      exact: current.sel.anchor.exact,
      prefix: current.sel.anchor.prefix,
      suffix: current.sel.anchor.suffix,
      decorations: [{ kind: 'bg', color }],
      createdAt: Date.now(),
    };
    const cls = `hl hl-${color}`;
    wrapRange(container2, haystack, current.sel.start, current.sel.end, cls, record.id);
    appliedRef.current.set(record.id, { start: current.sel.start, end: current.sel.end });
    void addHighlight(path, record).catch(() => showToast('高亮保存失败', { kind: 'error' }));
    window.getSelection()?.removeAllRanges();
    return;
  }
}, [menu, path]);

// 渲染处（ready 分支之后任意位置）：
{menu && (
  <SelectionMenu
    x={menu.x} y={menu.y}
    actions={menuActionsFor(menu.ctx)}
    onAction={onMenuAction}
    onClose={() => setMenu(null)}
  />
)}
// .doc-content 上： onContextMenu={onContentContextMenu}
```

```ts
export type OrientationState = 'portrait' | 'landscape' | 'landscape-flipped';

export const DEFAULT_ORIENTATION: OrientationState = 'portrait';
export const ORIENTATION_KEY = 'orientation';

export const LOCK_TARGET: Record<OrientationState, OrientationLockType> = {
  portrait: 'portrait',
  landscape: 'landscape-primary',
  'landscape-flipped': 'landscape-secondary',
};

export function sanitizeOrientation(raw: unknown): OrientationState {
  return raw === 'portrait' || raw === 'landscape' || raw === 'landscape-flipped' ? raw : DEFAULT_ORIENTATION;
}

/** 尝试系统级方向锁定；不支持（桌面/浏览器标签未全屏）或失败返回 false。 */
export async function applyOrientation(state: OrientationState): Promise<boolean> {
  try {
    if (typeof screen === 'undefined' || !screen.orientation || typeof screen.orientation.lock !== 'function') return false;
    await screen.orientation.lock(LOCK_TARGET[state]);
    return true;
  } catch {
    return false;
  }
}
```

- [ ] **Step 3: 实现 `src/state/OrientationContext.tsx`**（与 TokenContext 同模式）

```tsx
import { createContext, useCallback, useContext, useEffect, useRef, useState, type ReactNode } from 'react';
import { kvGet, kvSet } from '../data/db';
import { showToast } from '../lib/toast';
import { DEFAULT_ORIENTATION, ORIENTATION_KEY, applyOrientation, sanitizeOrientation, type OrientationState } from '../lib/orientation';

interface OrientationApi {
  state: OrientationState;
  setOrientation(next: OrientationState): void;
  toggleLandscape(): void;
  flip(): void;
}

const OrientationContext = createContext<OrientationApi | null>(null);

export function OrientationProvider({ children }: { children: ReactNode }) {
  const [state, setState] = useState<OrientationState>(DEFAULT_ORIENTATION);
  const [loaded, setLoaded] = useState(false);
  const lastLandscape = useRef<OrientationState>('landscape');

  useEffect(() => {
    let cancelled = false;
    void kvGet<unknown>(ORIENTATION_KEY).then(
      (v) => { if (!cancelled) { setState(sanitizeOrientation(v)); setLoaded(true); } },
      () => { if (!cancelled) setLoaded(true); },
    );
    return () => { cancelled = true; };
  }, []);

  // 启动时应用一次（静默：失败不打扰，用户操作时才会提示）。
  useEffect(() => {
    if (!loaded) return;
    void applyOrientation(state);
    // eslint-disable-next-line react-hooks/exhaustive-deps
  }, [loaded]);

  useEffect(() => {
    if (state !== 'portrait') lastLandscape.current = state;
  }, [state]);

  const setOrientation = useCallback((next: OrientationState) => {
    setState(next);
    void kvSet(ORIENTATION_KEY, next).catch(() => {});
    void applyOrientation(next).then((ok) => {
      if (!ok) showToast('当前环境不支持方向锁定：请用主屏幕应用（PWA）打开', { kind: 'error' });
    });
  }, []);

  const toggleLandscape = useCallback(() => {
    setOrientation(state === 'portrait' ? lastLandscape.current : 'portrait');
  }, [state, setOrientation]);

  const flip = useCallback(() => {
    if (state === 'landscape') setOrientation('landscape-flipped');
    else if (state === 'landscape-flipped') setOrientation('landscape');
  }, [state, setOrientation]);

  return (
    <OrientationContext.Provider value={{ state, setOrientation, toggleLandscape, flip }}>
      {children}
    </OrientationContext.Provider>
  );
}

export function useOrientation(): OrientationApi {
  const ctx = useContext(OrientationContext);
  if (!ctx) throw new Error('useOrientation 必须在 OrientationProvider 内使用');
  return ctx;
}
```
